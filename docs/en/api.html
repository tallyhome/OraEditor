Constructor

const editor = new OraEditor({
  element: "#editor",
  content: "<p>Hello</p>",
  toolbar: true,
  preset: "full",
  uploadImage: async (file, ctx) => ({ url, alt }),
  aiProxyUrl: "/ora-editor/ai",
});

getJSON / setJSON, getHTML / setHTML, runAI, exec, undo / redo, destroy.
